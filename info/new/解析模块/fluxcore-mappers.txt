<?xml version="1.0" encoding="UTF-8" ?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.platform.fluxcore.dao.pub.SystemConfigDao">
    <resultMap id="configMap" type="com.platform.fluxcore.entity.SystemConfig">
        <id column="id" property="id"/>
        <result column="config_key" property="configKey"/>
        <result column="config_value" property="configValue"/>
        <result column="description" property="description"/>
        <result column="create_time" property="createTime"/>
        <result column="update_time" property="updateTime"/>
        <result column="status" property="status"/>
    </resultMap>
    
    <select id="findAll" resultMap="configMap">
        SELECT * FROM tb_system_config WHERE status = 1
    </select>
    
    <select id="findByKey" resultMap="configMap">
        SELECT * FROM tb_system_config WHERE config_key = #{configKey} AND status = 1
    </select>
    
    <select id="findByPrefix" resultMap="configMap">
        SELECT * FROM tb_system_config WHERE config_key LIKE CONCAT(#{prefix}, '%') AND status = 1
    </select>
    
    <insert id="insert" parameterType="com.platform.fluxcore.entity.SystemConfig" useGeneratedKeys="true" keyProperty="id">
        INSERT INTO tb_system_config (
            config_key, config_value, description, 
            create_time, update_time, status
        ) VALUES (
            #{configKey}, #{configValue}, #{description}, 
            #{createTime}, #{updateTime}, #{status}
        )
    </insert>
    
    <update id="update" parameterType="com.platform.fluxcore.entity.SystemConfig">
        UPDATE tb_system_config SET 
            config_value = #{configValue},
            description = #{description},
            update_time = #{updateTime}
        WHERE config_key = #{configKey}
    </update>
    
    <update id="updateValue">
        UPDATE tb_system_config SET 
            config_value = #{configValue},
            update_time = NOW()
        WHERE config_key = #{configKey}
    </update>
    
    <delete id="deleteByKey">
        UPDATE tb_system_config SET status = 0 WHERE config_key = #{configKey}
    </delete>
</mapper>
