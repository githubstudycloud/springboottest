# 1. mongos.conf (路由节点配置)
systemLog:
  destination: file
  path: "/var/log/mongodb/mongos.log"
  logAppend: true

net:
  port: 27017
  maxIncomingConnections: 1000  # 路由节点连接数适中

security:
  keyFile: "/etc/mongodb/keyfile"

sharding:
  configDB: configReplSet/config1:27019,config2:27019,config3:27019

setParameter:
  maxWriteConflicts: 500       # 写入冲突限制
  cursorTimeoutMillis: 300000  # 游标超时时间
  internalQueryExecMaxBlockingSortBytes: 67108864  # 排序内存限制

# 2. mongod-config.conf (配置服务器配置)
storage:
  dbPath: "/var/lib/mongodb"
  wiredTiger:
    engineConfig:
      cacheSizeGB: 1           # 配置服务器缓存较小

systemLog:
  destination: file
  path: "/var/log/mongodb/mongod-config.log"
  logAppend: true

net:
  port: 27019
  maxIncomingConnections: 200  # 配置服务器连接数较少

replication:
  replSetName: "configReplSet"

sharding:
  clusterRole: configsvr

security:
  keyFile: "/etc/mongodb/keyfile"
  authorization: enabled

setParameter:
  maxWriteConflicts: 100       # 配置服务器写入冲突限制较小

# 3. mongod-shard.conf (分片节点配置)
storage:
  dbPath: "/var/lib/mongodb"
  wiredTiger:
    engineConfig:
      cacheSizeGB: 2          # 一半内存给缓存
    collectionConfig:
      blockCompressor: snappy # 使用snappy压缩
    indexConfig:
      prefixCompression: true # 启用索引前缀压缩

systemLog:
  destination: file
  path: "/var/log/mongodb/mongod-shard.log"
  logAppend: true

net:
  port: 27018
  maxIncomingConnections: 500 # 分片节点连接数适中

replication:
  replSetName: "shardReplSet"
  oplogSizeMB: 2048          # oplog大小2GB

sharding:
  clusterRole: shardsvr

security:
  keyFile: "/etc/mongodb/keyfile"
  authorization: enabled

setParameter:
  maxWriteConflicts: 300      # 分片节点写入冲突限制
  cursorTimeoutMillis: 300000
  wiredTigerConcurrentReadTransactions: 128  # 读取并发限制
  wiredTigerConcurrentWriteTransactions: 128 # 写入并发限制

operationProfiling:
  slowOpThresholdMs: 100      # 慢查询阈值

processManagement:
  fork: true
  pidFilePath: "/var/run/mongodb/mongod-shard.pid"
