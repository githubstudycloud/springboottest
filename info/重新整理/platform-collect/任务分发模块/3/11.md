


让我详细讲解如何使用这个JSON转Python类的工具。我会通过几个具体的例子来说明：

1. 基本用法 - 单个JSON转换：
```python
from json_to_class import ClassGenerator

# 创建生成器实例
generator = ClassGenerator()

# 示例JSON
json_str = '''
{
    "name": "张三",
    "age": 25,
    "address": {
        "city": "北京",
        "street": "朝阳区"
    }
}
'''

# 生成Python类代码
generated_code = generator.generate_code(json_str, "Person")

# 执行生成的代码
exec(generated_code)

# 使用生成的类
data = json.loads(json_str)
person = Person.from_dict(data)

# 访问对象属性
print(person.name)  # 输出: 张三
print(person.address.city)  # 输出: 北京
```

2. 处理包含null值的JSON：
```python
json_with_null = '''
{
    "name": "李四",
    "age": null,
    "email": "lisi@example.com",
    "address": null
}
'''

# 生成代码
generated_code = generator.generate_code(json_with_null, "Person")
exec(generated_code)

# 创建对象 - 支持空值
person = Person.from_dict(json.loads(json_with_null))
print(person.age)  # 输出: None
```

3. 处理JSON数组：
```python
json_array = '''
[
    {
        "id": 1,
        "name": "产品A",
        "price": 99.9
    },
    {
        "id": 2,
        "name": "产品B",
        "price": 199.9,
        "description": "新产品"  # 注意这个额外的字段
    }
]
'''

# 生成代码 - 会自动处理所有字段
generated_code = generator.generate_code(json_array, "Product")
exec(generated_code)

# 解析数组中的所有对象
products = []
data = json.loads(json_array)
for item in data:
    products.append(Product.from_dict(item))

# 访问对象
print(products[0].name)  # 输出: 产品A
print(products[1].description)  # 可以访问可选字段
```

4. 更新现有类定义：
```python
# 第一个JSON生成初始类
initial_json = '{"name": "王五", "age": 30}'
initial_code = generator.generate_code(initial_json, "Person")

# 新的JSON包含额外字段
new_json = '{"name": "赵六", "age": 25, "phone": "13800138000"}'

# 更新类定义
updated_code = generator.update_from_new_json(initial_code, new_json, "Person")
exec(updated_code)

# 现在Person类同时支持两种JSON格式
person1 = Person.from_dict(json.loads(initial_json))
person2 = Person.from_dict(json.loads(new_json))
```

5. 合并对象数据：
```python
# 两个不同的数据对象
data1 = {"name": "张三", "age": 20}
data2 = {"name": "李四", "phone": "13900139000"}

person1 = Person.from_dict(data1)
person2 = Person.from_dict(data2)

# 合并对象（将person2的非空字段合并到person1）
person1.merge(person2)
print(person1.to_dict())  
# 输出: {"name": "李四", "age": 20, "phone": "13900139000"}
```

2. 错误处理：
```python
try:
    generated_code = generator.generate_code(json_str)
except ValueError as e:
    print(f"JSON解析错误: {e}")
except Exception as e:
    print(f"生成代码错误: {e}")
```

3. 保存生成的代码：
