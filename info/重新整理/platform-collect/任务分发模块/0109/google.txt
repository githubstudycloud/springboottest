import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;
import java.util.List;
import java.util.ArrayList;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;

public class BatchTaskExecutor {

    private final ThreadPoolTaskExecutor executor;
    private final AtomicInteger runningTasks = new AtomicInteger(0);

    public BatchTaskExecutor(int corePoolSize, int maxPoolSize, int queueCapacity, boolean useVirtualThreads) {
        this.executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(corePoolSize);
        executor.setMaxPoolSize(maxPoolSize);
        executor.setQueueCapacity(queueCapacity);
        if (useVirtualThreads) {
            executor.setTaskExecutor(Executors.newVirtualThreadPerTaskExecutor()); // 使用虚拟线程
        }
        executor.initialize();
    }

    public <T> void executeBatches(List<T> tasks, int batchSize, BatchTask<T> batchTask) {
        int totalTasks = tasks.size();
        int numBatches = (int) Math.ceil((double) totalTasks / batchSize);

        for (int i = 0; i < numBatches; i++) {
            int start = i * batchSize;
            int end = Math.min((i + 1) * batchSize, totalTasks);
            List<T> batch = tasks.subList(start, end);

            CountDownLatch latch = new CountDownLatch(batch.size());

            for (T task : batch) {
                runningTasks.incrementAndGet();
                executor.submit(() -> {
                    long startTime = System.currentTimeMillis();
                    try {
                        batchTask.execute(task);
                    } finally {
                        long endTime = System.currentTimeMillis();
                        System.out.println("Task finished in " + (endTime - startTime) + "ms. Running tasks: " + runningTasks.decrementAndGet());
                        latch.countDown();
                    }
                });
            }
          try {
            latch.await(); // 等待当前批次完成
            System.out.println("Batch " + (i + 1) + " completed.");
          } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
          }
        }
        System.out.println("All batches completed.");
    }

    public void setMaxThreads(int maxThreads) {
        executor.setMaxPoolSize(maxThreads);
    }

    public interface BatchTask<T> {
        void execute(T task);
    }

    public static void main(String[] args) {
        // 示例用法
        BatchTaskExecutor executor = new BatchTaskExecutor(2, 10, 1000, true); // 使用虚拟线程
        List<Integer> numbers = new ArrayList<>();
        for (int i = 0; i < 100; i++) {
            numbers.add(i);
        }

        executor.executeBatches(numbers, 10, number -> {
            // 这里执行你的任务逻辑
            System.out.println("Processing number: " + number + " in thread " + Thread.currentThread().getName());
            try {
                Thread.sleep(100); // 模拟任务执行时间
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        });
        executor.setMaxThreads(5); // 动态调整线程数
        executor.executeBatches(numbers, 5, number -> {
            System.out.println("Processing number: " + number + " in thread " + Thread.currentThread().getName());
            try {
                Thread.sleep(100); // 模拟任务执行时间
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        });

    }
}

    private final BatchTaskConfig config;

    public BatchTaskExecutor(BatchTaskConfig config) {
        this.config = config;
        this.executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(config.corePoolSize);
        executor.setMaxPoolSize(config.maxPoolSize);
        executor.setQueueCapacity(config.queueCapacity);
        if (config.useVirtualThreads) {
            executor.setTaskExecutor(Executors.newVirtualThreadPerTaskExecutor());
        }
        executor.setThreadNamePrefix(config.threadNamePrefix);
        executor.initialize();
    }

    public <T, P> void executeBatchesWithNesting(String version, List<T> tasks, int batchSize, BatchTask<T, P> batchTask, P params) {
        if (runningVersions.putIfAbsent(version, true) != null) {
            config.logConsumer.accept("WARN", "Version " + version + " is already running.");
            return;
        }

        try {
            List<List<T>> pages = PageUtils.splitListIntoPages(tasks, batchSize);
            if (pages.isEmpty()) {
                config.logConsumer.accept("INFO", "Version " + version + " no data to process.");
                return;
            }
            CountDownLatch mainLatch = new CountDownLatch(pages.size());

            config.logConsumer.accept("INFO", "Version " + version + " start process, total pages: " + pages.size());
            long startTime = System.currentTimeMillis();

            for (List<T> page : pages) {
                runningTasks.incrementAndGet();
                executor.submit(() -> {
                    long taskStartTime = System.currentTimeMillis();
                    try {
                        batchTask.execute(page, params, this);
                    } catch (Throwable e) {
                        config.logConsumer.accept("ERROR", "Version " + version + " task execution error: " + e.getMessage());
                        config.exceptionHandler.accept(e); // 使用配置的异常处理器
                    } finally {
                        long taskEndTime = System.currentTimeMillis();
                        config.logConsumer.accept("INFO", "Version " + version + " Task finished in " + (taskEndTime - taskStartTime) + "ms. Running tasks: " + runningTasks.decrementAndGet());
                        mainLatch.countDown();
                    }
                });
            }
            mainLatch.await();
            long endTime = System.currentTimeMillis();
            config.logConsumer.accept("INFO", "Version " + version + " completed, total time: " + (endTime - startTime) + "ms.");
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            config.logConsumer.accept("ERROR", "Version " + version + " interrupted: " + e.getMessage());
            config.exceptionHandler.accept(e);
        } finally {
            runningVersions.remove(version);
        }
    }

    public <T, P> void submitTask(Runnable task) {
        executor.submit(task);
    }

    public void setMaxThreads(int maxThreads) {
        executor.setMaxPoolSize(maxThreads);
    }

    public interface BatchTask<T, P> {
        void execute(List<T> tasks, P params, BatchTaskExecutor executor) throws Exception;
    }

    public static void main(String[] args) throws InterruptedException {
        BatchTaskConfig config = new BatchTaskConfig()
                .setCorePoolSize(4)
                .setMaxPoolSize(20)
                .setQueueCapacity(2000)
                .setUseVirtualThreads(true)
                .setThreadNamePrefix("my-batch-task-")
                .setLogConsumer((level, message) -> System.out.println(String.format("[%s] %s %s", level, Thread.currentThread().getName(), message)))
                .setExceptionHandler(e -> {
                    System.err.println("Global Exception Handler catch exception:");
                    e.printStackTrace();
                });

        BatchTaskExecutor executor = new BatchTaskExecutor(config);
        List<String> versions = List.of("v1", "v2", "v3");
        for (String version : versions) {
            List<Integer> data = new ArrayList<>();
            for (int i = 0; i < 1000; i++) data.add(i);
            executor.executeBatchesWithNesting(version, data, 50, (tasks, versionInner, batchTaskExecutor) -> {
                System.out.println("Version :" + versionInner + " outer task: " + tasks.size() + " Thread: " + Thread.currentThread().getName());
                List<List<Integer>> pages = PageUtils.splitListIntoPages(tasks, 10);
                CountDownLatch innerLatch = new CountDownLatch(pages.size());
                for (List<Integer> page : pages) {
                    batchTaskExecutor.submitTask(() -> {
                        System.out.println("Version :" + versionInner + " inner task: " + page.size() + " Thread: " + Thread.currentThread().getName());
                        try {
                            Thread.sleep(100);
                        } catch (InterruptedException e) {
                            throw new RuntimeException(e);
                        }
                        innerLatch.countDown();
                    });
                }
                innerLatch.await();
            }, version);
        }
        Thread.sleep(1000);
        System.out.println("main thread end");
    }
}
