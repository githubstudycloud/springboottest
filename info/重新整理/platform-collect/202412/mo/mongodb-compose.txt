version: '3.8'

services:
  mongo1:
    image: mongo:4.4
    container_name: mongo1
    command: mongod --replSet rs0 --bind_ip_all --port 27017 --keyFile /data/mongodb.key
    ports:
      - "27017:27017"
    volumes:
      - ./data/mongo1:/data/db
      - ./mongod.key:/data/mongodb.key
    environment:
      MONGO_INITDB_ROOT_USERNAME: admin
      MONGO_INITDB_ROOT_PASSWORD: password123
    networks:
      - mongo-network
    deploy:
      resources:
        limits:
          memory: 1G
    restart: always
